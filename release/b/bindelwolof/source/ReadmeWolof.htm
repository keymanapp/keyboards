بِسْمِ اللّٰهِ الرَّحْمَـٰنِ الرَّحِيمِ

.1اُبْبِتۭ ڄِهْ دٜݣُبْ بِندْۨ ؤلٔڢْ:

دٜݣُ بِندْۨ بِي کُرۭيلُݣْٰ مَڢَاتِحُ الْبِشْرِ ݣَارَلْ، مُهْ دِهْ جُمْتُکَايُ خَرَلَ  ݣُهْ مَنَ بِندْۨ فَصِيدَ اَکْ لَٓکْکُ ؤلٔڢْ ڄِهْ اَجَمِيُ، اَجَمِيُ نَکْ مٔويْ حَرَڢِ اَلْفُرْءَانْ يِهْ جُؐهْ جٜٓکْۨکَۨلْ بٔولٜ ڄِهْ اَيْ ڄٔمْبَۢجْؐ اَکْ خَمْتُ يِهْ مَݣْ جِؐهْ دَانْ جٚڢٚدِۨکٗو؞ ݣۭنْنۭ نَهْ جُؐهْ ڄِهْ بَاتْ يِهْ اَکْ مَاسْکَ يِهْ لَٓککْ وِهْ سٔخْلَؤو کِۨرْ اَارْ  سُنُ ڄٔسَانْ اَکْ اَکْ سُنُ دِينۭ وَايٜ اِتَمْ يٜيسَلْ بِنْدِۨينُ اَجَمِيُ ڄِهْ رۭيوْ مِهْ اَکْ بِتِمْ رۭيوْ؞

.2 ݣِسْݣِسْ بِهْ اَکْ لَيْ يِهْ:

لُهْ اٚݑݑْ ڢُکْکِ خَرْنُ اَکْ جَؐارَ  کِۨهْ نِي ئوهْ خَمْ نٜهْ بِۢنْدُۨمْ اَجَمِيُ مٔوهْ اٚݑْݑٔنْ اَبْ تَخَوَايْ ڄِهْ جَٓلْلَلٜ خَمْخَمْ اَکْ دِينۭ ڄِهْ سُنُ رۭيوْ يِي؞ بَهْ تَخْ دٜݣُبْ بِندْۨ ؤلٔڢْ بِهْ دَيْ سَاڢَرَ جَۨٓمْبَۢتْلُهْ يَٓݣݣْ لُهْ جٚمْ ڄِهْ جُمْتَکَايُ خَرَلَاݣْ نُمٜرِکْ ݣُهْ مَنٜ اَکْ Windows ، Web،iOS، Androidبَهْ نٔݑْݑِ نَکْ دٚݑْݑٗوکْ  Unicode کَۨهْ خَمْ نٜهْ دۭيݣُبْ خَرَلَ بُهْ جۭلْلُمْ رۭيوْ لَهْ تٜهْ آَدْدُنٚ سۭݑݑْ بٔکْکُ کٔهْ؞

.3جُبْلُݣْ دٜݣُ بِندْۨ بِهْ:

٭يۭوۭنٚلْ اَکْ يَاتَلْ بِۢنْدُۨمْ ؤلٔڢْ ڄِهْ اَجَمِيُ

٭تَخَوَلْ لُݣٚتٗو ݣُهْ سَخْ دَٓککْ ڄِهْ حَرَڢْ يِهْ اَکْ خَمْتُ يِهْ؛

٭جۭبْبٚلٜ جُمْتُکَايْ ݣُهْ جۭؐݑݑْ مَنَ جٚڢٚدِۨکٗو؛

٭وٗورْلُ وَجْؐجَؐاتُݣْ بِۢندْۨ مِهْ اَکْ تُڄْڄُتُ ݣُهْ لٜيرْ؞

.4دٔخِينَمْ يِهْ ݣٚنٚ رَٓجْؐجِؐکُ:

◄ بٔکْکُ نَهْ ڄِهْ بِندْۨ بُهْ بَٓيْيِکٔو دٜۨيْجٔورْ جٚمْ ڄَامٔوجْؐ(RTL)؛

◄حَرَڢْ يِهْ اَکْ مَاسْکَ يِهْ اَکْ خَمْتُ يِهْ لۭݑݑْ دِهْ اَجَمِيُ؛

◄حَرَڢِ تٔݣْݣَلٜييْ بَاتْ اَکِيْٰ خَمْتُ؛

◄دٚݑْݑٗو نَهْ اَکْ Keyman Desktop اَکْ  Mobileاَکْ Web

.5جٚڢٚدِۨکُوِينْ وِهْ:

●بَسْ حَرَڢْ يِهْ ڄِهْ دٜݣُ بِهْ،

●دٔلْلِ ڄِهْ مَاسْکَ ݣِهْ،

●سُهْ کٔهْ دٜڢٜي خَمْتُ يِهْ دَهْ جُؐيْ تَبْبُ ڄِهْ سَاهْ سِهْ؛

●مَنْ نَهْ جؐٔوهْ جٚڢٚدِۨکٗو (Shift)کِۨرْ اَمْ يٜننْ خَمْتُ يِهْ؛

.6دۭۨکْؐکَانْ يُهْ اَمْ جَۨرِجْؐ:

۞ بِهْ ڄِهْ جۨٚککْ مٔويْ بِندْۨ اَوْ لَٓککْ دَهْ ڢَهْ لَاجْ دۭݣْݣٗو اَکُݣْ مَٓکْؐکٔو دَۨخْ جِؐهْ کٔهْ دٜڢْ دَهْ جؐٔوهْ اَمْ يۭينۭ رٜککٜ بَهْ تَخْ بُهْ جُؐهْ کٔهْ ڢٔٓکْؐکٜي، لُهْ ڄِهْ جَرَ سٔݑْݑِ مَنٜيسْ کٔو آَندْۨ سُݑْݑِ؛

۞ دَهْ ڢَهْ مٜلْ نِهْ کِۨرْ ئمْبَۢلْ مٔکْکَلْ ݣِهْ دِهْ نَهْ جَرْ ڢُتْتِکُ ڄِݣْ نَارْ لٔولُ مٔويْ  جُؐهْ بَجَؐ دٚݣٚرْ ڄِهْ لُيْ يٜيخَلْ سُنُݣْٰ دۭݣݣْ؛

۞جَؐتْتٜيلْ بِهْ مٔويْ بَاتْ بٔوي بِندْۨ رٜککْ دَيْ اٚݑݑْ وٜننْ حَرَڢْ، لُهْ بَاتْ ݣَٓتتْ ݣَٓتتْ مَتْ جَؐارِ حَرَڢْ؛

۞جٜؐينْتٜيلْ بِهْ مٔويْ بُهْ جَؐارِ مَاسْکَ نۭيݣْ وَرٜي جِيتْلٜ دَهْ جُؐيْ ڢَرْ ݣُهْ جۨٚککْ ݣِهْ؛

۞جُرٗومۭيلْ بِهْ مٔويْ، تٜهْ مٔوهْ وَرٔنَ جِيتُ مُيْ ڢٜخٜ بَهْ بَاتْ يِهْ جُؐيْ بِندْۨ دٔݣُ، بَاتْ بُهْ نٜککْ جُؐهْ خَمْ ڢُهْ مُهْ وَرَ يٜمْ جُؐهْ يَمَلٜ کٔهْ ڢَهْ؞

.7دۭيݣِ خَرَلَ يِهْ مُهْ دۭٓݣْݣٗولْ:

(Word, Google, LibreOffice…) Web ؛ iOS؛ Android ؛Windows

.8  حَٓففْ ڄِهْ جٚڢٚدِۨکُ ݣِهْ:

مَڢَاتُحُ الْبِشْرِ مٔوهْ کٔهْ جۭبْبٚلۭ کِۨرْ جَٓݑْݑُلٜ جَۨٓکْۨکَۨلٜ مِهْ اَکْ  جَٓلْلَلٜ خَمْخَمْ بِهْ اَکْ ݣٚسْتُ بِهْ ڄِهْ دِينۭ اَکْ ڄَادَ، کۭݑݑْ کُهْ کٔيْ جٚڢٚدۨٔکٔو بُلْ ڢَٓتْتٜ جِؐهْ کٔهْ سٔسْ، جَؐانَلْ لۭينْ کۨٚرٚمُلْ يَٓلْلَهْ؞

.9جٔکْکٔو اَکْ جٔلْلَسُ:

۞ مَڢَاتِحُ الْبِشْرِ طُوبَۍٰ دَارُ خَدِيمْ، ݑِکِنْ ݣِنَاوْ رَايْ دَکَارْ

بِسْمِ اللّٰهِ الرَّحْمَـٰنِ الرَّحِيمِ

 

.1اُبْبِتۭ ڄِهْ دٜݣُبْٰ بِندْۨ ؤلٔڢْ:

دٜݣُ بِندْۨ ؤلٔفْ بِي کُرۭيلُݣْٰ مَڢَاتِحُ الْبِشْرِݣَارَلْ، مُهْ دِهْ جُمْتُکَايُ خَرَلَ ݣُهْ مَنَ بِندْۨ فَصِيدَ اَکْ لَٓکْکُ ؤلٔڢْ ڄِهْ اَجَمِيُ، اَجَمِيُ نَکْ مٔويْ حَرَڢِ اَلْفُرْءَانْ يِهْ جُؐهْ جٜٓکْۨکَۨلْ بٔولٜ ڄِهْ اَيْ ڄٔمْبَۢجْؐ اَکْ خَمْتُ يِهْ مَݣْ جِؐهْ دَانْ جٚڢٚدِۨکٗو؞ ݣۭنْنۭ نَهْ جُؐهْ ڄِهْ بَاتْ يِهْ اَکْ مَاسْکَ يِهْ لَٓککْ وِهْ سٔخْلَؤو کِۨرْ اَارْ  سُنُ ڄٔسَانْ اَکْ سُنُ دِينۭ وَايٜ اِتَمْ يٜيسَلْ بِنْدِۨينُ اَجَمِيُ، بَهْ نٔݑْݑِ نَکْ دٚݑْݑٗوکْ Unicode مِهْ کَۨهْ خَمْ نٜهْ دۭيݣُبْٰ خَرَلَ بُهْ جۭلْلُمْٰ رۭيوْ لَهْ تٜهْ آَدْدُنٚ سۭݑݑْ بٔکْکُ کٔهْ؞
